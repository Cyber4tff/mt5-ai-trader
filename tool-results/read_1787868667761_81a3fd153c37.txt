    89→// ─── Store Interface ──────────────────────────────────────
    90→
    91→interface TradingStore {
    92→  // State
    93→  backendAvailable: boolean;
    94→  backendChecking: boolean;
    95→  connection: ConnectionState;
    96→  liveState: LiveConnectionState;
    97→  positions: Position[];
    98→  pendingOrders: Position[];
    99→  scanResults: ScanResult[];
   100→  riskSummary: DailyRiskSummary;
   101→  aiStatus: AIStatus;
   102→  autoTrade: AutoTradeState;
   103→  scanning: boolean;
   104→  scanLog: string[];
   105→  fetchingAccount: boolean;
   106→
   107→  // Actions
   108→  connectLive: (brokerName: string, brokerId: string, mt5Server: string) => Promise<void>;
   109→  startPaperTrading: (balance?: number, leverage?: number) => Promise<boolean>;
   110→  connect: (broker: string, login: number, password: string, server?: string) => Promise<boolean>;
   111→  disconnect: () => Promise<void>;
   112→  confirmMT5WithCredentials: (login: number, password: string, server: string) => Promise<boolean>;
   113→  unconfirmMT5Connection: () => void;
   114→  fetchMT5Account: () => Promise<void>;
   115→  fetchAccount: () => Promise<void>;
   116→  fetchRiskStatus: () => Promise<void>;
   117→  fetchAIStatus: () => Promise<void>;
   118→  scanMarkets: (symbols?: string[]) => Promise<void>;
   119→  toggleAutoTrade: (enabled: boolean, interval?: number) => Promise<void>;
   120→  closePosition: (ticket: number) => Promise<void>;
   121→  checkBackendHealth: () => Promise<boolean>;
   122→}
   123→
   124→function log(state: TradingStore, msg: string): string[] {
   125→  const ts = new Date().toLocaleTimeString();
   126→  return [...state.scanLog.slice(-200), `[${ts}] ${msg}`];
   127→}
   128→
   129→function parsePosition(p: Record<string, unknown>): Position {
   130→  return {
   131→    ticket: p.ticket as number,
   132→    symbol: (p.symbol as string) || "",
   133→    type: ((p.type as string) || "BUY").toUpperCase() as "BUY" | "SELL",
   134→    volume: p.volume as number,
   135→    open_price: p.open_price as number,
   136→    current_price: p.current_price as number,
   137→    sl: p.sl as number,
   138→    tp: p.tp as number,
   139→    profit: p.profit as number,
   140→    swap: p.swap as number,
   141→    comment: (p.comment as string) || "",
   142→    time: p.time as string,
   143→  };
   144→}

   589→        equity: data.equity as number,

   597→      const rawPositions = (data.positions as Record<string, unknown>[]) || [];
   598→      const rawOrders = (data.orders as Record<string, unknown>[]) || [];

   601→        connection: {
   602→          ...state.connection,

   671→  // ── Scan Markets (real analysis from backend) ─────────────────────────────────────────────
   672→  scanMarkets: async (symbols) => {
   673→    const state = get();
   674→    // Use the analysis session ID (works for both paper and live modes)
   675→    const sessionId = state.connection.analysisSessionId || state.connection.sessionId;
   676→    if (!sessionId) return;
   677→
   678→    set({ scanning: true });
   679→    set((s) => ({ scanLog: log(s, `Scanning ${symbols?.join(", ") || "all symbols"}...`) }));
   680→
   681→    try {
   682→      const data = await tradingApi.scan(
   683→        sessionId,
   684→        { symbols }
   685→      );
   686→      const rawResults = (data.results as Record<string, unknown>[]) || [];
   687→
   688→      const results: ScanResult[] = rawResults.map((r) => {
   689→        const actionable = r.actionable_signal
   690→          ? {
   691→              symbol: (r.actionable_signal as Record<string, unknown>).symbol as string,
   692→              direction: (r.actionable_signal as Record<string, unknown>).direction as "BUY" | "SELL" | "NO_TRADE",
   693→              entry: (r.actionable_signal as Record<string, unknown>).entry as number,
   694→              sl: (r.actionable_signal as Record<string, unknown>).sl as number,
   695→              tp: (r.actionable_signal as Record<string, unknown>).tp as number,
   696→              volume: (r.actionable_signal as Record<string, unknown>).volume as number,
   697→              confidence: (r.actionable_signal as Record<string, unknown>).confidence as number,
   698→              risk_reward: (r.actionable_signal as Record<string, unknown>).risk_reward as number,
   699→              confirmation_factors: ((r.actionable_signal as Record<string, unknown>).confirmation_factors as string[]) || [],
   700→              confluence: ((r.actionable_signal as Record<string, unknown>).confluence as ScanResult["confluence"]) || null,
   701→            }
   702→          : null;
   703→
   704→        return {
   705→          symbol: r.symbol as string,
   706→          confluence: (r.confluence as ScanResult["confluence"]) || null,
   707→          decisions_count: (r.decisions_count as number) || 0,
   708→          actionable,
   709→          errors: (r.errors as string[]) || [],
   710→          risk_failures: (r.risk_failures as string[]) || undefined,
   711→          timeframes: (r.timeframes as ScanResult["timeframes"]) || {},
   712→        };
   713→      });
   714→
   715→      const actionableCount = results.filter((r) => r.actionable).length;
   716→      const noTradeCount = results.length - actionableCount;
   717→
   718→      set((s) => ({
   719→        scanResults: results,
   720→        scanning: false,
   721→        autoTrade: { ...s.autoTrade, lastScan: new Date().toISOString(), cycleCount: s.autoTrade.cycleCount + 1 },
   722→        scanLog: [
   723→          ...log(s, `Scan complete: ${actionableCount} actionable, ${noTradeCount} NO TRADE`),
   724→          ...results.flatMap((r) => {
   725→            const msgs: string[] = [];
   726→            if (r.actionable) {
   727→              msgs.push(`${r.symbol}: ACTIONABLE ${r.actionable.direction} conf=${r.actionable.confidence.toFixed(2)} R:R=${r.actionable.risk_reward.toFixed(1)}`);
   728→            } else if (r.risk_failures?.length) {
   729→              msgs.push(`${r.symbol}: Signal rejected → ${r.risk_failures[0]}`);
   730→            } else if (r.errors?.length) {
   731→              msgs.push(`${r.symbol}: Error → ${r.errors[0]}`);
   732→            } else {
   733→              msgs.push(`${r.symbol}: NO TRADE`);
   734→            }
   735→            return msgs;
   736→          }),
   737→        ],
   738→      }));
   739→
   740→      get().fetchRiskStatus();
   741→    } catch (error) {
   742→      const msg = error instanceof Error ? error.message : "Unknown error";
   743→      set((s) => ({
   744→        scanning: false,
   745→        scanLog: log(s, `Scan error: ${msg}`),
   746→      }));
   747→    }
   748→  },
