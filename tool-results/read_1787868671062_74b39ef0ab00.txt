   124→   124→function log(state: TradingStore, msg: string): string[] {
   125→   125→  const ts = new Date().toLocaleTimeString();
   126→   126→  return [...state.scanLog.slice(-200), `[${ts}] ${msg}`];

   130→   130→  return {
   131→   131→    ticket: p.ticket as number,
   132→   132→    symbol: (p.symbol as string) || "",
   133→   133→    type: ((p.type as string) || "BUY").toUpperCase() as "BUY" | "SELL",
   134→   134→    volume: p.volume as number,
   135→   135→    open_price: p.open_price as number,
   136→   136→    current_price: p.current_price as number,
   137→   137→    sl: p.sl as number,
   138→   138→    tp: p.tp as number,
   139→   139→    profit: p.profit as number,
   140→   140→    swap: p.swap as number,
   141→   141→    comment: (p.comment as string) || "",
   142→   142→    time: p.time as string,
   143→   143→  };
   144→   144→}

   672→   672→  scanMarkets: async (symbols) => {
   673→   673→    const state = get();
   674→   674→    // Use the analysis session ID (works for both paper and live modes)
   675→   675→    const sessionId = state.connection.analysisSessionId || state.connection.sessionId;

   677→   677→
   678→   678→    set({ scanning: true });

   680→   680→
   681→   681→    try {
   682→   682→      const data = await tradingApi.scan(
   683→   683→        sessionId,
   684→   684→        { symbols }
   685→   685→      );
   686→   686→      const rawResults = (data.results as Record<string, unknown>[]) || [];
   687→   687→
   688→   688→      const results: ScanResult[] = rawResults.map((r) => {
   689→   689→        const actionable = r.actionable_signal
   690→   690→          ? {
   691→   691→              symbol: (r.actionable_signal as Record<string, unknown>).symbol as string,
   692→   692→              direction: (r.actionable_signal as Record<string, unknown>).direction as "BUY" | "SELL" | "NO_TRADE",
   693→   693→              entry: (r.actionable_signal as Record<string, unknown>).entry as number,
   694→   694→              sl: (r.actionable_signal as Record<string, unknown>).sl as number,
   695→   695→              tp: (r.actionable_signal as Record<string, unknown>).tp as number,
   696→   696→              volume: (r.actionable_signal as Record<string, unknown>).volume as number,
   697→   697→              confidence: (r.actionable_signal as Record<string, unknown>).confidence as number,
   698→   698→              risk_reward: (r.actionable_signal as Record<string, unknown>).risk_reward as number,
   699→   699→              confirmation_factors: ((r.actionable_signal as Record<string, unknown>).confirmation_factors as string[]) || [],
   700→   700→              confluence: ((r.actionable_signal as Record<string, unknown>).confluence as ScanResult["confluence"]) || null,
   701→   701→            }
   702→   702→          : null;
   703→   703→
   704→   704→        return {
   705→   705→          symbol: r.symbol as string,
   706→   706→          confluence: (r.confluence as ScanResult["confluence"]) || null,
   707→   707→          decisions_count: (r.decisions_count as number) || 0,
   708→   708→          actionable,
   709→   709→          errors: (r.errors as string[]) || [],
   710→   710→          risk_failures: (r.risk_failures as string[]) || undefined,
   711→   711→          timeframes: (r.timeframes as ScanResult["timeframes"]) || {},
   712→   712→        };
   713→   713→      });
   714→   714→
   715→   715→      const actionableCount = results.filter((r) => r.actionable).length;
   716→   716→      const noTradeCount = results.length - actionableCount;
   717→   717→
   718→   718→      set((s) => ({
   719→   719→        scanResults: results,
   720→   720→        scanning: false,
   721→   721→        autoTrade: { ...s.autoTrade, lastScan: new Date().toISOString(), cycleCount: s.autoTrade.cycleCount + 1 },
   722→   722→        scanLog: [
   723→   723→          ...log(s, `Scan complete: ${actionableCount} actionable, ${noTradeCount} NO TRADE`),
   724→   724→          ...results.flatMap((r) => {
   725→   725→            const msgs: string[] = [];
   726→   726→            if (r.actionable) {
   727→   727→              msgs.push(`${r.symbol}: ACTIONABLE ${r.actionable.direction} conf=${r.actionable.confidence.toFixed(2)} R:R=${r.actionable.risk_reward.toFixed(1)}`);
   728→   728→            } else if (r.risk_failures?.length) {
   729→   729→              msgs.push(`${r.symbol}: Signal rejected → ${r.risk_failures[0]}`);
   730→   730→            } else if (r.errors?.length) {
   731→   731→              msgs.push(`${r.symbol}: Error → ${r.errors[0]}`);
   732→   732→            } else {
   733→   733→              msgs.push(`${r.symbol}: NO TRADE`);
   734→   734→            }
   735→   735→            return msgs;
   736→   736→          }),
   737→   737→        ],
   738→   738→      }));
   739→   739→
   740→   740→      get().fetchRiskStatus();
   741→   741→    } catch (error) {
   742→   742→      const msg = error instanceof Error ? error.message : "Unknown error";
   743→   743→      set((s) => ({
   744→   744→        scanning: false,
   745→   745→        scanLog: log(s, `Scan error: ${msg}`),
