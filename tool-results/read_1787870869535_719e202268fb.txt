   108→    """Create a paper trading session."""
   109→    try:
   110→        broker_name = "Paper Trading"

   117→            broker=broker_name,

   123→        )
   124→
   125→        account = session.get_account()

   128→        for sym in settings.DEFAULT_SYMBOLS:

   135→            "session_id": session.session_id,
   136→            "broker": session.broker,
   137→            "server": session.server,

   159→def _connect_mt5(login: int, password: str, server: str) -> dict:
   160→    """Connect to MT5 terminal and return account info."""
   161→    global _mt5_connected, _mt5_account_info
   162→
   163→    if not MT5_AVAILABLE:
   164→        return {
   165→            "success": False,
   166→            "error": "MT5 library not available on this server. MetaTrader5 requires Windows with the MT5 desktop application installed.",
   167→            "mt5_available": False,
   168→        }
   169→
   170→    try:
   171→        if not mt5.initialize():
   172→            return {"success": False, "error": f"MT5 initialize failed: {mt5.last_error()}", "mt5_available": True}
   173→
   174→        authorized = mt5.login(login, password, server)
   175→        if not authorized:
   176→            error_code = mt5.last_error()
   177→            mt5.shutdown()
   178→            error_messages = {
   179→                1: "No connection with trade server",
   180→                2: "Invalid authorization",
   181→                3: "Invalid password",
   182→                10: "No connection with trade server",
   183→            }
   184→            msg = error_messages.get(error_code, f"Login failed (error {error_code})")
   185→            return {"success": False, "error": msg, "mt5_available": True, "error_code": error_code}
   186→
   187→        _mt5_connected = True
   188→        account = mt5.account_info()
   189→        _mt5_account_info = account
   190→
   191→        return {
   192→            "success": True,
   193→            "mt5_available": True,
   194→            "account": {
   195→                "balance": account.balance,
   196→                "equity": account.equity,
   197→                "margin": account.margin,
   198→                "free_margin": account.margin_free,
   199→                "leverage": account.leverage,
   200→                "profit": account.profit,
   201→                "margin_level": account.margin_level if account.margin > 0 else 0,
   202→                "login": account.login,
   203→                "name": account.name,
   204→                "server": account.server,
   205→                "currency": account.currency,
   206→            },
   207→        }
   208→    except Exception as e:
   209→        _mt5_connected = False
   210→        return {"success": False, "error": str(e), "mt5_available": True}

   346→def _execute_mt5_trade(symbol: str, direction: str, volume: float, sl: float, tp: float, comment: str) -> dict:
   347→    """Execute a market order on MT5."""
   348→    global _mt5_connected
   349→
   350→    if not MT5_AVAILABLE:
   351→        return {
   352→            "success": False,
   353→            "error": "MT5 library not available on this server. Trade execution requires Windows with the MT5 desktop application installed.",
   354→            "mt5_available": False,
   355→        }
   356→
   357→    if not _mt5_connected:
   358→        return {"success": False, "error": "Not connected to MT5. Please reconnect."}
   359→
   360→    try:
   361→        # Get symbol info for tick and volume constraints
   362→        symbol_info = mt5.symbol_info(symbol)
   363→        if symbol_info is None:
   364→            return {"success": False, "error": f"Symbol {symbol} not found on broker"}
   365→
   366→        if not symbol_info.visible:
   367→            if not mt5.symbol_select(symbol, True):
   368→                return {"success": False, "error": f"Cannot select symbol {symbol}"}
   369→            symbol_info = mt5.symbol_info(symbol)
   370→
   371→        tick = mt5.symbol_info_tick(symbol)
   372→        if tick is None:
   373→            return {"success": False, "error": f"Cannot get tick data for {symbol}"}
   374→
   375→        # Normalize volume to broker's step size
   376→        volume_step = symbol_info.volume_step
   377→        min_lot = symbol_info.volume_min
   378→        max_lot = symbol_info.volume_max
   379→
   380→        volume = round(round(volume / volume_step) * volume_step, 8)
   381→        volume = max(min_lot, min(max_lot, volume))
   382→
   383→        # Determine order type and price
   384→        if direction.upper() == "BUY":
   385→            order_type = mt5.ORDER_TYPE_BUY
   386→            price = tick.ask
   387→        elif direction.upper() == "SELL":
   388→            order_type = mt5.ORDER_TYPE_SELL
   389→            price = tick.bid
   390→        else:
   391→            return {"success": False, "error": f"Invalid direction: {direction}. Use BUY or SELL."}
   392→
   393→        # Determine filling mode
   394→        filling_type = symbol_info.filling_mode
   395→        if filling_type == 1:
   396→            type_filling = mt5.ORDER_FILLING_FOK
   397→        elif filling_type == 2:
   398→            type_filling = mt5.ORDER_FILLING_IOC
   399→        else:
   400→            type_filling = mt5.ORDER_FILLING_RETURN
   401→
   402→        request = {
   403→            "action": mt5.TRADE_ACTION_DEAL,
   404→            "symbol": symbol,
   405→            "volume": volume,
   406→            "type": order_type,
   407→            "price": price,
   408→            "sl": sl,
   409→            "tp": tp,
   410→            "deviation": 20,
   411→            "magic": 234000,
   412→            "comment": comment or "AI Cloud Trader",
   413→            "type_time": mt5.ORDER_TIME_GTC,
   414→            "type_filling": type_filling,
   415→        }
   416→
   417→        print(f"[MT5 Trade] Sending: {symbol} {direction} {volume} lots @ {price} SL={sl} TP={tp}")
   418→        result = mt5.order_send(request)
   419→
   420→        if result is None:
   421→            error_code = mt5.last_error()
   422→            error_msg = _mt5_error_message(error_code)
   423→            return {"success": False, "error": f"Order send failed: {error_msg}", "error_code": error_code}
   424→
   425→        if result.retcode != mt5.TRADE_RETCODE_DONE:
   426→            error_msg = _mt5_error_message(result.retcode)
   427→            return {
   428→                "success": False,
   429→                "error": error_msg,
   430→                "retcode": result.retcode,
   431→                "deal": result.deal,
   432→                "order": result.order,
   433→            }
   434→
   435→        # Get the opened position
   436→        position = None
   437→        if result.deal > 0:
   438→            positions = mt5.positions_get(ticket=result.order)
   439→            if positions and len(positions) > 0:
   440→                p = positions[0]
   441→                position = {
   442→                    "ticket": p.ticket,
   443→                    "symbol": p.symbol,
   444→                    "type": "BUY" if p.type == 0 else "SELL",
   445→                    "volume": p.volume,
   446→                    "open_price": p.price_open,
   447→                    "current_price": p.price_current,
   448→                    "sl": p.sl,
   449→                    "tp": p.tp,
   450→                    "profit": 0.0,
   451→                    "swap": 0.0,
   452→                    "comment": p.comment,
   453→                    "time": str(p.time),
   454→                }
   455→
   456→        print(f"[MT5 Trade] SUCCESS: Deal #{result.deal} Order #{result.order} {symbol} {direction} {volume} @ {price}")
   457→
   458→        return {
   459→            "success": True,
   460→            "deal": result.deal,
   461→            "order": result.order,
   462→            "price": result.price,
   463→            "volume": result.volume,
   464→            "comment": result.comment,
   465→            "position": position,
   466→        }
   467→
   468→    except Exception as e:
   469→        traceback.print_exc()
   470→        return {"success": False, "error": str(e)}

   473→def _close_mt5_position(ticket: int) -> dict:
   474→    """Close an MT5 position by ticket."""
   475→    if not MT5_AVAILABLE or not _mt5_connected:
   476→        return {"success": False, "error": "Not connected to MT5"}
   477→
   478→    try:
   479→        positions = mt5.positions_get(ticket=ticket)
   480→        if not positions or len(positions) == 0:
   481→            return {"success": False, "error": f"Position #{ticket} not found"}
   482→
   483→        pos = positions[0]
   484→        tick = mt5.symbol_info_tick(pos.symbol)
   485→        if tick is None:
   486→            return {"success": False, "error": f"Cannot get tick for {pos.symbol}"}
   487→
   488→        # Close with opposite direction
   489→        if pos.type == 0:  # BUY
   490→            close_type = mt5.ORDER_TYPE_SELL
   491→            price = tick.bid
   492→        else:  # SELL
   493→            close_type = mt5.ORDER_TYPE_BUY
   494→            price = tick.ask
   495→
   496→        symbol_info = mt5.symbol_info(pos.symbol)
   497→        filling_type = symbol_info.filling_mode
   498→        if filling_type == 1:
   499→            type_filling = mt5.ORDER_FILLING_FOK
   500→        elif filling_type == 2:
   501→            type_filling = mt5.ORDER_FILLING_IOC
   502→        else:
   503→            type_filling = mt5.ORDER_FILLING_RETURN
   504→
   505→        request = {
   506→            "action": mt5.TRADE_ACTION_DEAL,
   507→            "symbol": pos.symbol,
   508→            "volume": pos.volume,
   509→            "type": close_type,
   510→            "position": pos.ticket,
   511→            "price": price,
   512→            "deviation": 20,
   513→            "magic": 234000,
   514→            "comment": "Close AI Trade",
   515→            "type_time": mt5.ORDER_TIME_GTC,
   516→            "type_filling": type_filling,
   517→        }
   518→
   519→        print(f"[MT5 Close] Closing position #{ticket} {pos.symbol} {pos.volume} lots @ {price}")
   520→        result = mt5.order_send(request)
   521→
   522→        if result is None or result.retcode != mt5.TRADE_RETCODE_DONE:
   523→            error_code = result.retcode if result else mt5.last_error()
   524→            return {"success": False, "error": _mt5_error_message(error_code), "retcode": error_code}
   525→
   526→        print(f"[MT5 Close] SUCCESS: Deal #{result.deal} closed position #{ticket}")
   527→        return {
   528→            "success": True,
   529→            "deal": result.deal,
   530→            "order": result.order,
   531→            "price": result.price,
   532→            "profit": pos.profit,
   533→        }
   534→
   535→    except Exception as e:
   536→        traceback.print_exc()
   537→        return {"success": False, "error": str(e)}

   666→
   667→def _analyze_symbol(symbol: str, session: TradingSession) -> Dict:
   668→    """Analyze a single symbol across all timeframes."""
   669→    tf_data: Dict[str, pd.DataFrame] = {}
   670→    analyses: Dict[str, MarketAnalysis] = {}
   671→    timeframe_results: Dict = {}
   672→
   673→    # Fetch candles for each timeframe
   674→    for interval in settings.MTF_TIMEFRAMES:
   675→        df = fetch_candles(symbol, interval)
   676→        if df is not None:
   677→            tf_data[interval] = df
   678→
   679→    # Analyze each available timeframe
   680→    for interval, df in tf_data.items():
   681→        try:
   682→            analysis = multi_tf.analyze_timeframe(df, symbol, interval)
   683→            analyses[interval] = analysis
   684→
   685→            # Map interval to display key (e.g. "1d" -> "D1")
   686→            tf_display = {"1d": "D1", "4h": "H4", "1h": "H1", "15m": "M15", "5m": "M5"}.get(interval, interval.upper())
   687→            bos_events = [b for b in analysis.structure_breaks if b.type == "BOS"]
   688→            choch_events = [b for b in analysis.structure_breaks if b.type == "CHOCH"]
   689→
   690→            timeframe_results[tf_display] = {
   691→                "trend": analysis.trend.value,
   692→                "bias": analysis.bias.value,
   693→                "atr": round(analysis.atr, 5) if analysis.atr else 0,
   694→                "volatility": analysis.volatility_regime or "normal",
   695→                "momentum": analysis.momentum or "neutral",
   696→                "bos": len(bos_events),
   697→                "choch": len(choch_events),
   698→                "signals_count": len(analysis.signals),
   699→                "structure_breaks": len(bos_events) + len(choch_events),
   700→                "sr_levels": len(analysis.sr_levels),
   701→            }
   702→        except Exception as e:
   703→            tf_display = {"1d": "D1", "4h": "H4", "1h": "H1", "15m": "M15", "5m": "M5"}.get(interval, interval.upper())
   704→            timeframe_results[tf_display] = {"error": str(e)}
   705→
   706→    # Compute confluence across all timeframes
   707→    confluence = multi_tf.compute_confluence(analyses)
   708→
   709→    # Find best signal across all timeframes
   710→    best_signal = None
   711→    for tf, analysis in analyses.items():
   712→        if analysis.signals:
   713→            for sig in analysis.signals:
   714→                if best_signal is None or sig.confidence > best_signal.confidence:
   715→                    best_signal = sig
   716→
   717→    # Run AI decision engine
   718→    actionable = None
   719→    risk_failures = None
   720→    decisions_count = 0
   721→
   722→    if best_signal and best_signal.signal_type != SignalType.NO_TRADE:
   723→        decision = ai_engine.evaluate(best_signal, confluence, session)
   724→        decisions_count = 1
   725→
   726→        if decision["all_checks_passed"]:
   727→            # Try to open the position
   728→            direction = decision["direction"]
   729→            pos = trading_engine.open_position(
   730→                session=session,
   731→                symbol=symbol,
   732→                direction=direction,
   733→                entry=best_signal.entry_price,
   734→                sl=best_signal.stop_loss,
   735→                tp=best_signal.take_profit,
   736→                confidence=best_signal.confidence,
   737→                reason=best_signal.reason,
   738→            )
   739→
   740→            if pos:
   741→                actionable = {
   742→                    "symbol": symbol,
   743→                    "direction": direction,
   744→                    "entry": pos.open_price,
   745→                    "sl": pos.sl,
   746→                    "tp": pos.tp,
   747→                    "volume": pos.volume,
   748→                    "confidence": best_signal.confidence,
   749→                    "risk_reward": best_signal.risk_reward,
   750→                    "confirmation_factors": decision["confirmation_factors"],
   751→                    "confluence": {
   752→                        "direction": confluence["direction"],
   753→                        "score": confluence["score"],
   754→                        "trend_alignment": confluence["trend_alignment"],
   755→                        "factors": confluence["factors"],
   756→                    },
   757→                }
   758→            else:
   759→                risk_failures = ["Position could not be opened (risk limits)"]
   760→        else:
   761→            risk_failures = decision["rejection_reasons"]
   762→            decisions_count = 1
   763→
   764→    # Build confluence for response
   765→    confluence_resp = None
   766→    if confluence.get("factors") and confluence["factors"] != ["No data"]:
   767→        confluence_resp = {
   768→            "direction": confluence["direction"],
   769→            "score": round(confluence["score"], 2),
   770→            "trend_alignment": confluence["trend_alignment"],
   771→            "factors": confluence["factors"],
   772→            "higher_tf_bias": confluence["higher_tf_bias"].value,
   773→            "bullish_ratio": round(confluence["bullish_ratio"], 2),
   774→            "bearish_ratio": round(confluence["bearish_ratio"], 2),
   775→        }
   776→
   777→    return {
   778→        "symbol": symbol,
   779→        "confluence": confluence_resp,
   780→        "decisions_count": decisions_count,
   781→        "actionable": actionable,
   782→        "errors": [],
   783→        "risk_failures": risk_failures,
   784→        "timeframes": timeframe_results,
   785→    }

   887→async def _background_price_updater():
   888→    """Periodically update prices for all open positions."""
   889→    while True:
   890→        try:
   891→            for session_id, session in list(trading_engine.sessions.items()):
   892→                if session.positions:
   893→                    trading_engine.update_positions(session)
   894→        except Exception as e:
   895→            print(f"[Background] Price update error: {e}")
   896→        await asyncio.sleep(10)  # Update every 10 seconds
   897→

   899→@app.on_event("startup")
   900→async def startup():
   901→    global _price_update_task
   902→    _price_update_task = asyncio.create_task(_background_price_updater())
   903→    print("[Engine] Cloud Trading Engine started on port 8001")
   904→    print("[Engine] No MT5 dependency - uses real market data from Yahoo Finance")
   905→

   908→async def shutdown():
   909→    if _price_update_task:
